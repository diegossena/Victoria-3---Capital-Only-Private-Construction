on_monthly_pulse = {
	effects = {
		every_country = {
			limit = { is_player = yes }
			every_owned_state = {
				limit = { is_capital = no }
				if = {
					limit = { 
						building_exists = building_financial_district 
					}
					downsize = building_financial_district
				}
				if = {
					limit = { 
						building_exists = building_manor_house 
					}
					downsize = building_manor_house
				}
			}
		}
	}
}
